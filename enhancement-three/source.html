# Example Python Code to Insert a Document

from pymongo import MongoClient
from bson.objectid import ObjectId
import logging


# Use logging instead of print() for errors (more professional / industry standard).
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class AnimalShelter(object):
    """ CRUD operations for Animal collection in MongoDB """

    def __init__(self, username, password):
        """
        Initialize MongoDB connection.

        Parameters:
            username (str): MongoDB username (ex: 'aacuser')
            password (str): MongoDB password

        Notes:
            - Connects to the aac database and animals collection
            - Authenticates against the admin database (per SNHU examples)
        """
        # Initializing the MongoClient to access MongoDB
        # This connects to the aac database and animals collection
        # using the aacuser account
        USER = username
        PASS = password
        HOST = 'localhost'
        PORT = 27017
        DB = 'aac'
        COL = 'animals'

        # Validate credentials early so errors are clear and consistent.
        # (Industry standard: validate inputs before attempting network calls.)
        if not isinstance(USER, str) or not isinstance(PASS, str) or USER == "" or PASS == "":
            raise ValueError("MongoDB username and password must be non-empty strings.")

        try:
            # Create the MongoDB client connection
            self.client = MongoClient(
                'mongodb://%s:%s@%s:%d/admin' % (USER, PASS, HOST, PORT)
            )

            # Access the specified database and collection
            self.database = self.client['%s' % (DB)]
            self.collection = self.database['%s' % (COL)]

            # Optional connection check (fails fast if auth/connection is wrong)
            self.client.admin.command('ping')

        except Exception as e:
            # Log the error and re-raise so the calling code knows connection failed.
            logger.error("MongoDB connection failed: %s", e)
            raise

    # Create method to insert a document into the animals collection
    def create(self, data):
        """
        Insert one document into MongoDB.

        Parameters:
            data (dict): Document to insert.

        Returns:
            bool: True if insert succeeds, otherwise False.
        """
        # Industry standard: validate inputs and fail safely.
        if data is None or not isinstance(data, dict) or len(data) == 0:
            logger.warning("Create failed: data must be a non-empty dictionary.")
            return False

        try:
            # Insert the document into MongoDB
            self.collection.insert_one(data)
            return True

        except Exception as e:
            logger.error("Create failed: %s", e)
            return False

    # Read method to retrieve documents from the animals collection
    def read(self, query):
        """
        Retrieve documents using a MongoDB query.

        Parameters:
            query (dict): Query filter. Use {} to retrieve all.

        Returns:
            list: Matching documents (empty list if none found or error).
        """
        results = []

        # Treat None as "retrieve all" to avoid crashes and support edge-case testing.
        if query is None:
            query = {}

        # Validate query format
        if not isinstance(query, dict):
            logger.warning("Read failed: query must be a dictionary.")
            return results

        try:
            # Find matching documents using the query dictionary
            cursor = self.collection.find(query)
            for document in cursor:
                results.append(document)

        except Exception as e:
            logger.error("Read failed: %s", e)

        # Return a list of matching documents (or empty list if none found)
        return results

    # Update method to change document(s) in the animals collection
    def update(self, query, new_values):
        """
        Update matching document(s) using a query and MongoDB update operation.

        Parameters:
            query (dict): Query filter to match documents.
            new_values (dict): Update operation, ex: {"$set": {"color": "Black"}}

        Returns:
            int: Number of documents modified (0 if none modified or error).
        """
        if query is not None and new_values is not None:
            # Validate types for industry-standard robustness
            if not isinstance(query, dict) or not isinstance(new_values, dict):
                logger.warning("Update failed: query and new_values must be dictionaries.")
                return 0

            try:
                # Update matching document(s) using the query and update data
                # new_values must be in the format accepted by MongoDB, such as:
                # {"$set": {"color": "Black"}}
                result = self.collection.update_many(query, new_values)
                return result.modified_count

            except Exception as e:
                logger.error("Update failed: %s", e)
                return 0
        else:
            return 0

    # Delete method to remove document(s) from the animals collection
    def delete(self, query):
        """
        Delete matching document(s) using a MongoDB query.

        Parameters:
            query (dict): Query filter to match documents for deletion.

        Returns:
            int: Number of documents deleted (0 if none deleted or error).
        """
        if query is not None:
            # Validate query type (prevents common edge-case crashes)
            if not isinstance(query, dict):
                logger.warning("Delete failed: query must be a dictionary.")
                return 0

            try:
                # Delete matching document(s) using the query
                result = self.collection.delete_many(query)
                return result.deleted_count

            except Exception as e:
                logger.error("Delete failed: %s", e)
                return 0
        else:
            return 0

            lat = dff.iloc[row, 13]

# Grazioso Salvare Animal Rescue Dashboard Database Enhancement

## Purpose

This directory contains the enhanced version of Alondra Paulino Santos's CS 340
Project Two artifact for the CS 499 database category. The original files remain
in the parent project directory for before-and-after comparison.

The enhancement preserves the dashboard's rescue-filtering purpose while moving
filtering, sorting, pagination, counting, and breed aggregation into MongoDB.
Only one requested page is transferred to the interface, while the chart still
summarizes the complete matching result set.

## Enhanced files

- `animal_shelter.py` provides validated CRUD operations, deterministic database
  paging, aggregation, error separation, and index-analysis helpers.
- `dashboard.py` provides the enhanced Dash application and safe user-query
  translation.
- `rescue_queries.py` centralizes the approved rescue-category rules shared by
  the dashboard and performance-evaluation utility.
- `index_evaluation.py` records existing indexes and comparable execution
  statistics. It does not create the documented candidate index unless the
  `--create-candidate` flag is supplied.
- `assets/GraziosoSalvareLogo.png` provides the original course-project
  branding as a locally served dashboard asset.
- `.env.example` documents required configuration names without real secrets.
- `tests/` contains isolated unit tests that do not modify a live database.

## Security decisions

Real credentials are not stored in source code. The application accepts only
approved filter and sort fields, translates a limited set of comparison
operations, caps page sizes, rejects empty update/delete queries, logs technical
database failures, and provides safe messages to dashboard users.

## Configuration

Set the following environment variables before starting the dashboard:

```text
MONGODB_USERNAME
MONGODB_PASSWORD
MONGODB_HOST
MONGODB_PORT
MONGODB_DATABASE
MONGODB_COLLECTION
MONGODB_AUTH_SOURCE
```

Defaults for nonsecret settings appear in `.env.example`. Do not place a real
password in `.env.example`, the notebook, screenshots, recordings, or Git.

## Install and run

From the `enhanced` directory:

```bash
python -m pip install -r requirements.txt
python dashboard.py
```

Open the local URL shown by Dash. The dashboard presents the original Grazioso
Salvare course-project logo alongside a readable title and author line.

## Logo provenance

The Grazioso Salvare logo is the original image distributed for the SNHU CS 340
course artifact. The recovered PNG was verified against two independently
published copies of the same course project; both copies had the same filename,
dimensions, file size, and SHA-256 checksum. The asset is stored locally so the
dashboard does not contact a third-party server while running.

## Tests

Run the isolated test suite from the `enhanced` directory:

```bash
python -m unittest discover -s tests -v
```

The tests cover query boundaries, sort stability, page limiting, identifier
serialization, aggregation construction, destructive-operation safeguards,
filter parsing, and coordinate validation. A live MongoDB verification should
also be performed with the course dataset before final submission.

## Index evaluation

First record the baseline with the current database:

```bash
python index_evaluation.py --output index_baseline.json
```

Review the existing indexes and candidate trade-offs. If the candidate is
appropriate, create it and repeat the identical queries:

```bash
python index_evaluation.py --create-candidate --output index_candidate.json
```

Compare returned-record counts, documents examined, keys examined, and execution
time. Execution time alone can vary between runs, so conclusions should consider
all measurements and confirm that results remain correct. The candidate index
uses storage and adds maintenance cost to writes; it should be retained only when
the measured benefit justifies that cost.

## Live verification checklist

1. Start MongoDB with the Austin Animal Center dataset and a least-privilege user.
2. Verify All Animals and all three rescue categories.
3. Apply approved text and numeric column filters.
4. Sort by one and multiple columns and move through multiple pages.
5. Confirm that total counts and top-breed statistics use the same criteria.
6. Verify a successful empty result displays `No animals match`.
7. Stop MongoDB temporarily and confirm a separate safe failure message appears.
8. Select animals with valid, missing, and invalid coordinates.
9. Record baseline and candidate index evidence against the same dataset.

"""Secure MongoDB data-access services for the enhanced rescue dashboard.

Author: Alondra Paulino Santos
Course: CS 499 Computer Science Capstone
Artifact: CS 340 Grazioso Salvare Animal Rescue Dashboard
Enhancement category: Databases
Version: 2.0
Date: September 30, 2026

Intent and design decisions:
    This module keeps database responsibilities outside the Dash interface. It
    validates every dashboard-generated query, performs sorting and pagination
    in MongoDB, calculates breed summaries with an aggregation pipeline, and
    exposes index-analysis helpers. Credentials are supplied through environment
    variables by the application rather than embedded in source code.
"""

from __future__ import annotations

from dataclasses import dataclass
import logging
import math
from typing import Any, Iterable, Mapping, Sequence
from urllib.parse import quote_plus

from bson import ObjectId
from pymongo import ASCENDING, DESCENDING, MongoClient
from pymongo.collection import Collection
from pymongo.errors import PyMongoError


LOGGER = logging.getLogger(__name__)

MAX_PAGE_SIZE = 100
DEFAULT_PAGE_SIZE = 10

# The dashboard may query only these known dataset fields. Keeping this list at
# the data boundary prevents arbitrary field and operator injection.
ALLOWED_FILTER_FIELDS = {
    "animal_id",
    "animal_type",
    "breed",
    "color",
    "name",
    "outcome_type",
    "sex_upon_outcome",
    "age_upon_outcome_in_weeks",
}

ALLOWED_SORT_FIELDS = ALLOWED_FILTER_FIELDS | {"datetime", "monthyear", "_id"}
ALLOWED_COMPARISON_OPERATORS = {"$eq", "$ne", "$gt", "$gte", "$lt", "$lte", "$in"}

# A projection limits network transfer to information the enhanced dashboard
# displays or uses. MongoDB's _id remains available as a stable tie-breaker.
DASHBOARD_PROJECTION = {
    "animal_id": 1,
    "animal_type": 1,
    "breed": 1,
    "color": 1,
    "name": 1,
    "outcome_type": 1,
    "sex_upon_outcome": 1,
    "age_upon_outcome_in_weeks": 1,
    "location_lat": 1,
    "location_long": 1,
}


class DatabaseOperationError(RuntimeError):
    """Represent a database failure without exposing connection details to UI code."""


@dataclass(frozen=True)
class PageResult:
    """One database page plus the metadata required by the dashboard."""

    records: list[dict[str, Any]]
    total_count: int
    page_number: int
    page_size: int

    @property
    def page_count(self) -> int:
        """Return the number of available pages, including zero for no matches."""

        return math.ceil(self.total_count / self.page_size) if self.total_count else 0


def _require_non_empty_text(value: Any, name: str) -> str:
    """Validate required configuration while keeping messages free of secrets."""

    if not isinstance(value, str) or not value.strip():
        raise ValueError(f"{name} must be a non-empty string.")
    return value.strip()


def _validate_scalar(value: Any, field: str) -> Any:
    """Allow predictable BSON-safe scalar values for dashboard filters."""

    if value is None or isinstance(value, (str, int, float, bool)):
        return value
    raise ValueError(f"Unsupported value type for filter field '{field}'.")


def validate_query(query: Mapping[str, Any] | None) -> dict[str, Any]:
    """Return a defensive copy of an allow-listed dashboard query.

    Rescue rules and user filters are joined with ``$and``. Nested ``$and`` is
    therefore allowed, but user-supplied top-level database operators are not.
    Each field may contain a scalar equality value or a small, approved set of
    comparison operators.
    """

    if query is None:
        return {}
    if not isinstance(query, Mapping):
        raise ValueError("Query must be a mapping.")

    # Build a new dictionary instead of mutating or trusting the caller's data.
    # This ensures that only values examined by this function reach PyMongo.
    validated: dict[str, Any] = {}
    for field, condition in query.items():
        if field == "$and":
            # The dashboard joins the selected rescue category and optional
            # column filters with $and. Each nested part is validated again.
            if not isinstance(condition, Sequence) or isinstance(condition, (str, bytes)):
                raise ValueError("$and must contain a sequence of query mappings.")
            validated["$and"] = [validate_query(part) for part in condition]
            continue

        if field not in ALLOWED_FILTER_FIELDS:
            raise ValueError(f"Filtering by '{field}' is not permitted.")

        if isinstance(condition, Mapping):
            # A condition such as {"$gte": 26} must use an approved comparison
            # operator. Database scripting and raw regex operators are rejected.
            safe_condition: dict[str, Any] = {}
            for operator, value in condition.items():
                if operator not in ALLOWED_COMPARISON_OPERATORS:
                    raise ValueError(f"Operator '{operator}' is not permitted.")
                if operator == "$in":
                    if not isinstance(value, Sequence) or isinstance(value, (str, bytes)):
                        raise ValueError("$in requires a list of values.")
                    safe_condition[operator] = [_validate_scalar(item, field) for item in value]
                else:
                    safe_condition[operator] = _validate_scalar(value, field)
            validated[field] = safe_condition
        else:
            validated[field] = _validate_scalar(condition, field)
    return validated


def validate_sort(sort: Sequence[tuple[str, int]] | None) -> list[tuple[str, int]]:
    """Validate sort fields and append ``_id`` for deterministic pagination."""

    # Rebuild the sort specification so a browser request cannot inject an
    # arbitrary database field or unsupported sort value.
    normalized: list[tuple[str, int]] = []
    for field, direction in sort or []:
        if field not in ALLOWED_SORT_FIELDS:
            raise ValueError(f"Sorting by '{field}' is not permitted.")
        if direction not in (ASCENDING, DESCENDING):
            raise ValueError("Sort direction must be ascending or descending.")
        if field != "_id":
            normalized.append((field, direction))

    # Stable ordering prevents records with equal values from moving between
    # pages while a user browses the same result set.
    normalized.append(("_id", ASCENDING))
    return normalized


def serialize_document(document: Mapping[str, Any]) -> dict[str, Any]:
    """Convert BSON-only values to dashboard-safe representations."""

    serialized = dict(document)
    if "_id" in serialized:
        serialized["_id"] = str(serialized["_id"])
    return serialized


class AnimalShelter:
    """Provide validated CRUD, paging, aggregation, and index operations."""

    def __init__(
        self,
        username: str,
        password: str,
        host: str = "localhost",
        port: int = 27017,
        database_name: str = "aac",
        collection_name: str = "animals",
        auth_source: str = "admin",
        *,
        client: MongoClient | None = None,
        connect_timeout_ms: int = 5000,
    ) -> None:
        """Connect to MongoDB or accept an injected client for isolated tests."""

        username = _require_non_empty_text(username, "MongoDB username")
        password = _require_non_empty_text(password, "MongoDB password")
        host = _require_non_empty_text(host, "MongoDB host")
        database_name = _require_non_empty_text(database_name, "Database name")
        collection_name = _require_non_empty_text(collection_name, "Collection name")
        auth_source = _require_non_empty_text(auth_source, "Authentication source")
        if not isinstance(port, int) or not 1 <= port <= 65535:
            raise ValueError("MongoDB port must be between 1 and 65535.")

        try:
            if client is None:
                # quote_plus protects reserved URI characters in credentials.
                # The completed URI is never logged or returned to the browser.
                uri = (
                    f"mongodb://{quote_plus(username)}:{quote_plus(password)}@"
                    f"{host}:{port}/?authSource={quote_plus(auth_source)}"
                )
                client = MongoClient(uri, serverSelectionTimeoutMS=connect_timeout_ms)
            # Selecting the database and collection here keeps those connection
            # details out of the dashboard's presentation logic.
            self.client = client
            self.database = self.client[database_name]
            self.collection: Collection = self.database[collection_name]
            # Ping fails fast during startup instead of waiting for the first
            # user action to reveal an authentication or connectivity problem.
            self.client.admin.command("ping")
        except PyMongoError as exc:
            LOGGER.exception("MongoDB connection failed.")
            raise DatabaseOperationError("The database connection could not be established.") from exc

    def close(self) -> None:
        """Release the MongoDB client's network resources."""

        self.client.close()

    def create(self, data: Mapping[str, Any]) -> str:
        """Insert one non-empty document and return its identifier."""

        if not isinstance(data, Mapping) or not data:
            raise ValueError("Create data must be a non-empty mapping.")
        try:
            result = self.collection.insert_one(dict(data))
            return str(result.inserted_id)
        except PyMongoError as exc:
            LOGGER.exception("Create operation failed.")
            raise DatabaseOperationError("The record could not be created.") from exc

    def read(self, query: Mapping[str, Any] | None = None) -> list[dict[str, Any]]:
        """Read validated records for compatibility with the original artifact."""

        safe_query = validate_query(query)
        try:
            return [serialize_document(item) for item in self.collection.find(safe_query)]
        except PyMongoError as exc:
            LOGGER.exception("Read operation failed.")
            raise DatabaseOperationError("The records could not be loaded.") from exc

    def find_page(
        self,
        query: Mapping[str, Any] | None,
        *,
        page_number: int = 0,
        page_size: int = DEFAULT_PAGE_SIZE,
        sort: Sequence[tuple[str, int]] | None = None,
        projection: Mapping[str, int] | None = None,
    ) -> PageResult:
        """Return one deterministically sorted MongoDB page and total count."""

        if not isinstance(page_number, int) or page_number < 0:
            raise ValueError("Page number must be a nonnegative integer.")
        if not isinstance(page_size, int) or not 1 <= page_size <= MAX_PAGE_SIZE:
            raise ValueError(f"Page size must be between 1 and {MAX_PAGE_SIZE}.")

        # Validate every user-influenced value before constructing the cursor.
        safe_query = validate_query(query)
        safe_sort = validate_sort(sort)
        safe_projection = dict(projection or DASHBOARD_PROJECTION)
        safe_projection["_id"] = 1

        try:
            # Count uses the same filter as the page query. The interface uses
            # this number to calculate pages and describe the complete result.
            total_count = self.collection.count_documents(safe_query)
            page_count = math.ceil(total_count / page_size) if total_count else 0
            # A filter change can make the browser's previous page invalid.
            # Returning to page zero produces a useful, predictable result.
            safe_page = page_number if page_count == 0 or page_number < page_count else 0
            # MongoDB performs sorting, skipping, and limiting before documents
            # cross the network, so the browser receives one requested page.
            cursor = (
                self.collection.find(safe_query, safe_projection)
                .sort(safe_sort)
                .skip(safe_page * page_size)
                .limit(page_size)
            )
            records = [serialize_document(item) for item in cursor]
            return PageResult(records, total_count, safe_page, page_size)
        except PyMongoError as exc:
            LOGGER.exception("Paged read operation failed.")
            raise DatabaseOperationError("The requested page could not be loaded.") from exc

    def aggregate_breed_counts(
        self, query: Mapping[str, Any] | None, limit: int = 10
    ) -> list[dict[str, Any]]:
        """Summarize breeds across every matching record, not one table page."""

        if not isinstance(limit, int) or not 1 <= limit <= 50:
            raise ValueError("Aggregation limit must be between 1 and 50.")
        safe_query = validate_query(query)
        # This pipeline deliberately repeats the table's validated filter. It
        # summarizes all matching animals, not only the current table page.
        pipeline = [
            {"$match": safe_query},
            {"$match": {"breed": {"$ne": None}}},
            {"$group": {"_id": "$breed", "count": {"$sum": 1}}},
            {"$sort": {"count": -1, "_id": 1}},
            {"$limit": limit},
            {"$project": {"_id": 0, "breed": "$_id", "count": 1}},
        ]
        try:
            return list(self.collection.aggregate(pipeline))
        except PyMongoError as exc:
            LOGGER.exception("Breed aggregation failed.")
            raise DatabaseOperationError("Breed statistics could not be loaded.") from exc

    def update(self, query: Mapping[str, Any], new_values: Mapping[str, Any]) -> int:
        """Update matching records while rejecting an accidental update-all call."""

        safe_query = validate_query(query)
        if not safe_query:
            raise ValueError("Update requires a non-empty query.")
        if not isinstance(new_values, Mapping) or not new_values:
            raise ValueError("Update values must be a non-empty mapping.")
        # Restrict updates to the predictable $set operation. Replacement and
        # scripting-style updates are outside this artifact's requirements.
        if set(new_values) != {"$set"} or not isinstance(new_values["$set"], Mapping):
            raise ValueError("Only a non-empty $set update is permitted.")
        if not new_values["$set"]:
            raise ValueError("$set must include at least one field.")
        try:
            return self.collection.update_many(safe_query, dict(new_values)).modified_count
        except PyMongoError as exc:
            LOGGER.exception("Update operation failed.")
            raise DatabaseOperationError("The records could not be updated.") from exc

    def delete(self, query: Mapping[str, Any]) -> int:
        """Delete matching records while rejecting an accidental delete-all call."""

        safe_query = validate_query(query)
        if not safe_query:
            raise ValueError("Delete requires a non-empty query.")
        try:
            return self.collection.delete_many(safe_query).deleted_count
        except PyMongoError as exc:
            LOGGER.exception("Delete operation failed.")
            raise DatabaseOperationError("The records could not be deleted.") from exc

    def list_indexes(self) -> list[dict[str, Any]]:
        """Return current index metadata so evaluation starts from evidence."""

        try:
            return list(self.collection.list_indexes())
        except PyMongoError as exc:
            LOGGER.exception("Index inspection failed.")
            raise DatabaseOperationError("Indexes could not be inspected.") from exc

    def create_index(self, keys: Iterable[tuple[str, int]], name: str) -> str:
        """Create one reviewed compound index; this is never called automatically."""

        # Index creation is an explicit evaluation action, never an automatic
        # side effect of starting the public dashboard.
        key_list = list(keys)
        if not key_list:
            raise ValueError("An index requires at least one field.")
        for field, direction in key_list:
            if field not in ALLOWED_SORT_FIELDS or direction not in (ASCENDING, DESCENDING):
                raise ValueError("Index specification contains an unsupported field or direction.")
        try:
            return self.collection.create_index(key_list, name=name)
        except PyMongoError as exc:
            LOGGER.exception("Index creation failed.")
            raise DatabaseOperationError("The index could not be created.") from exc

    def explain_find(
        self,
        query: Mapping[str, Any] | None,
        sort: Sequence[tuple[str, int]] | None = None,
    ) -> dict[str, Any]:
        """Return execution statistics for repeatable before-and-after evaluation."""

        safe_query = validate_query(query)
        safe_sort = validate_sort(sort)
        try:
            # executionStats supplies returned, examined-document, examined-key,
            # and timing evidence for the same query before and after indexing.
            command = {
                "explain": {
                    "find": self.collection.name,
                    "filter": safe_query,
                    "sort": dict(safe_sort),
                    "limit": DEFAULT_PAGE_SIZE,
                },
                "verbosity": "executionStats",
            }
            return self.database.command(command)
        except PyMongoError as exc:
            LOGGER.exception("Query explanation failed.")
            raise DatabaseOperationError("Query statistics could not be collected.") from exc

def combine_queries(*queries: Mapping[str, Any]) -> dict[str, Any]:
    """Combine non-empty query parts without silently overwriting conditions."""

    # $and preserves conditions when the rescue rule and a user filter target
    # the same field; dictionary merging could silently overwrite one of them.
    parts = [dict(query) for query in queries if query]
    if not parts:
        return {}
    if len(parts) == 1:
        return parts[0]
    return {"$and": parts}


def _coerce_filter_value(field: str, raw_value: str) -> Any:
    """Convert approved numeric filter values while leaving text as text."""

    # Dash includes surrounding quotes for text filters. Removing only the
    # outer quote characters leaves the intended literal value for MongoDB.
    value = raw_value.strip().strip('"').strip("'")
    if len(value) > 100:
        raise ValueError("Filter values may not exceed 100 characters.")
    if field in NUMERIC_COLUMNS:
        try:
            return float(value)
        except ValueError as exc:
            raise ValueError(f"{field} requires a numeric value.") from exc
    return value


FILTER_EXPRESSION = re.compile(
    r"^\{(?P<field>[A-Za-z0-9_]+)\}\s*"
    r"(?P<operator>eq|ne|ge|le|gt|lt|=|!=|>=|<=|>|<)\s*"
    r"(?P<value>.+)$"
)

OPERATOR_MAP = {
    "eq": "$eq",
    "=": "$eq",
    "ne": "$ne",
    "!=": "$ne",
    "ge": "$gte",
    ">=": "$gte",
    "le": "$lte",
    "<=": "$lte",
    "gt": "$gt",
    ">": "$gt",
    "lt": "$lt",
    "<": "$lt",
}


def parse_table_filter(filter_query: str | None) -> dict[str, Any]:
    """Translate Dash filter expressions into an allow-listed MongoDB query.

    Dash joins column expressions with ``&&``. This parser intentionally
    supports comparison operations only. It does not accept raw JSON, regular
    expressions, JavaScript, or MongoDB operators from the browser.
    """

    if not filter_query or not filter_query.strip():
        return {}

    # Parse each expression individually instead of accepting browser-provided
    # JSON. That creates a narrow, explainable boundary for database access.
    conditions: list[dict[str, Any]] = []
    for expression in filter_query.split(" && "):
        match = FILTER_EXPRESSION.fullmatch(expression.strip())
        if not match:
            raise ValueError(
                "Use filters such as {breed} eq 'Labrador Retriever Mix' or "
                "{age_upon_outcome_in_weeks} ge 26."
            )
        # Field and operation checks happen before the value becomes a query.
        field = match.group("field")
        if field not in FILTERABLE_COLUMNS:
            raise ValueError(f"Filtering by '{field}' is not available.")
        value = _coerce_filter_value(field, match.group("value"))
        conditions.append({field: {OPERATOR_MAP[match.group("operator")]: value}})

    return conditions[0] if len(conditions) == 1 else {"$and": conditions}


def parse_sort(sort_by: list[dict[str, str]] | None) -> list[tuple[str, int]]:
    """Translate Dash sort selections into validated PyMongo directions."""

    # DataTable reports human-readable directions; PyMongo uses 1 and -1.
    parsed: list[tuple[str, int]] = []
    for item in sort_by or []:
        column = item.get("column_id")
        if column not in FILTERABLE_COLUMNS:
            raise ValueError(f"Sorting by '{column}' is not available.")
        direction = ASCENDING if item.get("direction") == "asc" else DESCENDING
        parsed.append((column, direction))
    return parsed


def valid_coordinates(record: Mapping[str, Any]) -> tuple[float, float] | None:
    """Return verified latitude and longitude or ``None`` for invalid data."""

    # Conversion rejects missing fields and nonnumeric strings before the map
    # component sees them.
    try:
        latitude = float(record["location_lat"])
        longitude = float(record["location_long"])
    except (KeyError, TypeError, ValueError):
        return None
    # Finite and geographic-range checks prevent NaN, infinity, or impossible
    # coordinates from producing a misleading marker.
    if not (math.isfinite(latitude) and math.isfinite(longitude)):
        return None
    if not (-90 <= latitude <= 90 and -180 <= longitude <= 180):
        return None
    return latitude, longitude

@callback(
    Output("datatable-id", "data"),
    Output("datatable-id", "page_count"),
    Output("datatable-id", "page_current"),
    Output("datatable-id", "selected_rows"),
    Output("breed-count-store", "data"),
    Output("status-message", "children"),
    Input("filter-type", "value"),
    Input("datatable-id", "filter_query"),
    Input("datatable-id", "sort_by"),
    Input("datatable-id", "page_current"),
    Input("datatable-id", "page_size"),
)
def update_dashboard(
    rescue_type: str,
    filter_query: str,
    sort_by: list[dict[str, str]],
    page_current: int,
    page_size: int,
):
    """Retrieve one page and matching aggregate statistics from MongoDB."""

    # A missing connection leaves the page usable and produces a safe status
    # message instead of allowing the callback to raise an uncaught exception.
    if database is None:
        return [], 0, 0, [], [], STARTUP_ERROR

    try:
        # Convert interface choices into one database query, then use that exact
        # query for the table page and the breed summary.
        rescue_query = RESCUE_QUERIES.get(rescue_type, {})
        user_query = parse_table_filter(filter_query)
        combined_query = combine_queries(rescue_query, user_query)
        # find_page performs count, sort, skip, and limit in MongoDB.
        result = database.find_page(
            combined_query,
            page_number=page_current or 0,
            page_size=page_size or DEFAULT_PAGE_SIZE,
            sort=parse_sort(sort_by),
        )
        breed_counts = database.aggregate_breed_counts(combined_query)
    except ValueError as exc:
        return [], 0, 0, [], [], f"Invalid filter or paging option: {exc}"
    except DatabaseOperationError:
        return [], 0, 0, [], [], "Animal records could not be loaded. Please try again."

    # A successful empty result is intentionally different from the database
    # failure handled above, giving the user an accurate explanation.
    if result.total_count == 0:
        return [], 0, 0, [], breed_counts, "No animals match the selected criteria."

    status = (
        f"Showing page {result.page_number + 1} of {result.page_count}; "
        f"{result.total_count:,} matching animals."
    )
    return (
        result.records,
        result.page_count,
        result.page_number,
        [0] if result.records else [],
        breed_counts,
        status,
    )

@callback(

    Input("datatable-id", "data"),
    Input("datatable-id", "selected_rows"),

def update_map(records: list[dict[str, Any]], selected_rows: list[int]):
    """Display the selected animal only when its coordinates are supported."""

    map_children: list[Any] = [dl.TileLayer(id="base-layer-id")]
    center = DEFAULT_CENTER
    message = "Select an animal with valid coordinates to display its location."

    if records:
        # Filtering or paging can invalidate an earlier selection. Defaulting
        # to row zero connects the marker to a record on the displayed page.
        row = selected_rows[0] if selected_rows and selected_rows[0] < len(records) else 0
        record = records[row]
        coordinates = valid_coordinates(record)
        if coordinates:
            # A marker is created only from coordinates validated above. The
            # application never invents a location for an incomplete record.
            latitude, longitude = coordinates
            center = [latitude, longitude]
            breed = str(record.get("breed") or "Unknown breed")
            name = str(record.get("name") or "Unnamed animal")
            map_children.append(
                dl.Marker(
                    position=center,
                    children=[dl.Tooltip(breed), dl.Popup([html.H4(name), html.P(breed)])],
                )
            )
            message = ""
        else:
            message = "The selected animal does not have valid map coordinates."

    return html.Div(

enhanced_artifact/index_evaluation.py
"""Collect repeatable MongoDB index evidence for the CS 499 enhancement.

Author: Alondra Paulino Santos
Version: 2.0
Date: September 30, 2026

This script never creates an index unless ``--create-candidate`` is supplied.
Run it once for a baseline, review the candidate against the actual dataset,
then run it with the flag and compare the generated execution statistics.
"""

from __future__ import annotations

import argparse
import json
import os
from pathlib import Path
from typing import Any

from pymongo import ASCENDING

from animal_shelter import AnimalShelter
from rescue_queries import RESCUE_QUERIES


CANDIDATE_INDEX = [
    ("animal_type", ASCENDING),
    ("sex_upon_outcome", ASCENDING),
    ("age_upon_outcome_in_weeks", ASCENDING),
    ("breed", ASCENDING),
]
CANDIDATE_NAME = "rescue_filter_compound_v1"


def _execution_summary(explanation: dict[str, Any]) -> dict[str, Any]:
    """Keep the comparable execution figures and omit verbose server details."""

    stats = explanation.get("executionStats", {})
    return {
        "n_returned": stats.get("nReturned"),
        "execution_time_ms": stats.get("executionTimeMillis"),
        "total_keys_examined": stats.get("totalKeysExamined"),
        "total_documents_examined": stats.get("totalDocsExamined"),
    }


def configured_database() -> AnimalShelter:
    """Build the same environment-configured connection used by the dashboard."""

    return AnimalShelter(
        username=os.environ["MONGODB_USERNAME"],
        password=os.environ["MONGODB_PASSWORD"],
        host=os.getenv("MONGODB_HOST", "localhost"),
        port=int(os.getenv("MONGODB_PORT", "27017")),
        database_name=os.getenv("MONGODB_DATABASE", "aac"),
        collection_name=os.getenv("MONGODB_COLLECTION", "animals"),
        auth_source=os.getenv("MONGODB_AUTH_SOURCE", "admin"),
    )


def main() -> None:
    """Inspect indexes, optionally create one candidate, and save comparable evidence."""

    parser = argparse.ArgumentParser(description="Evaluate a reviewed rescue-query index.")
    parser.add_argument(
        "--create-candidate",
        action="store_true",
        help="Create the documented compound-index candidate before collecting statistics.",
    )
    parser.add_argument(
        "--output",
        default="index_evaluation_results.json",
        help="Path for the evidence file.",
    )
    args = parser.parse_args()

    # Reuse the same configuration and data-access class as the dashboard so
    # the evaluation reflects the deployed collection and validation rules.
    database = configured_database()
    try:
        # Preserve the starting index inventory. This prevents an unsupported
        # claim that the original collection had no useful indexes.
        existing_before = database.list_indexes()
        if args.create_candidate:
            # Creation is intentionally opt-in after baseline evidence is saved.
            database.create_index(CANDIDATE_INDEX, CANDIDATE_NAME)

        evaluations = {}
        for category in ("water", "mountain", "disaster"):
            # All categories use the same sorting and limit so before-and-after
            # statistics remain comparable.
            explanation = database.explain_find(
                RESCUE_QUERIES[category],
                sort=[("age_upon_outcome_in_weeks", ASCENDING)],
            )
            evaluations[category] = _execution_summary(explanation)

        # The JSON output records measurements and the design trade-off together,
        # making the final performance claim auditable in the ePortfolio.
        evidence = {
            "candidate_created_for_this_run": args.create_candidate,
            "candidate_name": CANDIDATE_NAME,
            "candidate_fields": CANDIDATE_INDEX,
            "existing_indexes_before_run": existing_before,
            "existing_indexes_after_run": database.list_indexes(),
            "query_execution_statistics": evaluations,
            "trade_off_note": (
                "The candidate may reduce reads for rescue filters, but every additional "
                "index consumes storage and adds maintenance work to inserts and updates."
            ),
        }
        Path(args.output).write_text(json.dumps(evidence, indent=2, default=str), encoding="utf-8")
        print(f"Index evidence written to {args.output}")
    finally:
        database.close()


if __name__ == "__main__":
    main()

from __future__ import annotations

import unittest
from unittest.mock import MagicMock

from bson import ObjectId
from pymongo import ASCENDING, DESCENDING

from animal_shelter import (
    DASHBOARD_PROJECTION,
    AnimalShelter,
    PageResult,
    serialize_document,
    validate_query,
    validate_sort,
)


class AnimalShelterTests(unittest.TestCase):
    """Exercise enhancement logic without requiring a live MongoDB server."""

    def setUp(self) -> None:
        self.collection = MagicMock()
        self.database = MagicMock()
        self.service = AnimalShelter.__new__(AnimalShelter)
        self.service.collection = self.collection
        self.service.database = self.database
        self.service.client = MagicMock()

    def test_query_allows_expected_rescue_conditions(self) -> None:
        query = {
            "$and": [
                {"animal_type": "Dog"},
                {"breed": {"$in": ["Newfoundland", "Bloodhound"]}},
                {"age_upon_outcome_in_weeks": {"$gte": 26, "$lte": 156}},
            ]
        }
        self.assertEqual(validate_query(query), query)

    def test_query_rejects_unapproved_fields_and_operators(self) -> None:
        with self.assertRaises(ValueError):
            validate_query({"password": "secret"})
        with self.assertRaises(ValueError):
            validate_query({"breed": {"$where": "malicious"}})

    def test_sort_adds_unique_identifier_tie_breaker(self) -> None:
        self.assertEqual(
            validate_sort([("breed", DESCENDING)]),
            [("breed", DESCENDING), ("_id", ASCENDING)],
        )

    def test_find_page_limits_work_in_database_and_serializes_id(self) -> None:
        document_id = ObjectId()
        cursor = MagicMock()
        cursor.sort.return_value = cursor
        cursor.skip.return_value = cursor
        cursor.limit.return_value = [{"_id": document_id, "breed": "Newfoundland"}]
        self.collection.find.return_value = cursor
        self.collection.count_documents.return_value = 21

        result = self.service.find_page(
            {"animal_type": "Dog"}, page_number=1, page_size=10, sort=[("breed", ASCENDING)]
        )

        self.assertIsInstance(result, PageResult)
        self.assertEqual(result.total_count, 21)
        self.assertEqual(result.page_count, 3)
        self.assertEqual(result.page_number, 1)
        self.assertEqual(result.records[0]["_id"], str(document_id))
        cursor.skip.assert_called_once_with(10)
        cursor.limit.assert_called_once_with(10)
        projection = dict(DASHBOARD_PROJECTION)
        projection["_id"] = 1
        self.collection.find.assert_called_once_with({"animal_type": "Dog"}, projection)

    def test_invalid_page_returns_to_first_page(self) -> None:
        cursor = MagicMock()
        cursor.sort.return_value = cursor
        cursor.skip.return_value = cursor
        cursor.limit.return_value = []
        self.collection.find.return_value = cursor
        self.collection.count_documents.return_value = 5

        result = self.service.find_page({}, page_number=9, page_size=10)
        self.assertEqual(result.page_number, 0)
        cursor.skip.assert_called_once_with(0)

    def test_aggregation_uses_match_group_sort_limit_and_projection(self) -> None:
        expected = [{"breed": "Labrador Retriever Mix", "count": 12}]
        self.collection.aggregate.return_value = expected
        result = self.service.aggregate_breed_counts({"animal_type": "Dog"})
        self.assertEqual(result, expected)

        pipeline = self.collection.aggregate.call_args.args[0]
        self.assertEqual(pipeline[0], {"$match": {"animal_type": "Dog"}})
        self.assertIn("$group", pipeline[2])
        self.assertEqual(pipeline[4], {"$limit": 10})

    def test_update_and_delete_reject_empty_queries(self) -> None:
        with self.assertRaises(ValueError):
            self.service.update({}, {"$set": {"name": "Test"}})
        with self.assertRaises(ValueError):
            self.service.delete({})

    def test_serialize_document_preserves_identifier_as_text(self) -> None:
        document_id = ObjectId()
        result = serialize_document({"_id": document_id, "name": "Teddy"})
        self.assertEqual(result, {"_id": str(document_id), "name": "Teddy"})


if __name__ == "__main__":
