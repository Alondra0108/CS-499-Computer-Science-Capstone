// Domain Model
struct Course {
    string number;               // normalized (trimmed, uppercased) e.g., "CSCI200"
    string title;                // course title
    vector<string> prereqs;      // normalized prerequisite course numbers
};

// Utility: trimming & normalization
static inline string ltrim(const string& s) {
    size_t i = 0;
    while (i < s.size() && std::isspace(static_cast<unsigned char>(s[i]))) ++i;
    return s.substr(i);
}
static inline string rtrim(const string& s) {
    if (s.empty()) return s;
    size_t i = s.size();
    while (i > 0 && std::isspace(static_cast<unsigned char>(s[i - 1]))) --i;
    return s.substr(0, i);
}
static inline string trim(const string& s) { return rtrim(ltrim(s)); }

// Normalize course codes so comparisons are consistent (e.g., "csci200 " -> "CSCI200").
static inline string NormalizeCourse(const string& raw) {
    string t = trim(raw);
    for (char& c : t) c = static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
    return t;
}
/* Reviewer note (Normalization):
   Normalize all course codes (trim + uppercase) before any hashing/lookup,
   so user input like "csci 200" matches "CSCI200" in the data. Prevents subtle
   mismatches and makes the rest of the pipeline stable. */

   // -------------------------------
   // Hash Table (chaining)
   // -------------------------------
struct Node {
    Course data;
    Node* next = nullptr;
    explicit Node(const Course& c) : data(c) {}
};

class HashTable {
public:
    explicit HashTable(size_t tableSize = 179) : tableSize_(tableSize), buckets_(tableSize_, nullptr) {}

    // Insert returns false on duplicate course number; otherwise true.
    bool Insert(const Course& c) {
        size_t idx = Hash(c.number);
        // Check duplicate in chain
        for (Node* cur = buckets_[idx]; cur != nullptr; cur = cur->next) {
            if (cur->data.number == c.number) return false;
        }
        Node* n = new Node(c);
        n->next = buckets_[idx];
        buckets_[idx] = n;
        return true;
    }

    // Search returns pointer to Course if found; otherwise nullptr.
    const Course* Search(const string& courseNumber) const {
        string key = NormalizeCourse(courseNumber);
        size_t idx = Hash(key);
        for (Node* cur = buckets_[idx]; cur != nullptr; cur = cur->next) {
            if (cur->data.number == key) return &cur->data;
        }
        return nullptr;
    }

    // Gather all courses to a vector (no side effects on table).
    vector<Course> ToVector() const {
        vector<Course> out;
        out.reserve(256);
        for (Node* head : buckets_) {
            for (Node* cur = head; cur != nullptr; cur = cur->next) {
                out.push_back(cur->data);
            }
        }
        return out;
    }

    // Gather and return courses sorted alphanumerically by course number.
    vector<Course> ToVectorSorted() const {
        vector<Course> v = ToVector();

        // Adaptive choice: insertion sort for tiny sets, std::sort otherwise.
        if (v.size() < 50) {
            for (size_t i = 1; i < v.size(); ++i) {
                Course key = v[i];
                size_t j = i;
                while (j > 0 && v[j - 1].number > key.number) {
                    v[j] = v[j - 1];
                    --j;
                }
                v[j] = key;
            }
        }
        else {
            std::sort(v.begin(), v.end(), [](const Course& a, const Course& b) {
                return a.number < b.number;
                });
        }
        return v;
    }
    /* Reviewer note (Hash + Sorting):
       - Table uses chaining with a prime bucket count to keep average inserts/lookups ~O(1).
       - Sorting is adaptive: insertion sort for small N, std::sort for larger lists.
         This keeps the implementation simple but still responsive for typical inputs. */

    ~HashTable() {
        for (Node* head : buckets_) {
            while (head) {
                Node* nxt = head->next;
                delete head;
                head = nxt;
            }
        }
    }

private:
    // Simple 31-based rolling hash for strings; mod table size (prime recommended).
    size_t Hash(const string& key) const {
        long long sum = 0;
        for (unsigned char ch : key) sum = sum * 31 + ch;
        if (sum < 0) sum = -sum;
        return static_cast<size_t>(sum) % tableSize_;
    }

    size_t tableSize_;
    vector<Node*> buckets_;
};

// Load/Validation Reporting
struct LoadIssue {
    size_t lineNo{};
    string type;   // e.g., "MissingField", "Duplicate", "UnknownPrereq", "SelfPrereq", "Cycle"
    string detail; // human-readable
};

struct LoadResultSummary {
    size_t linesRead = 0;
    size_t parsedCourses = 0;
    size_t inserted = 0;
    size_t duplicates = 0;
    size_t unknownPrereqs = 0;
    size_t selfPrereqs = 0;
    size_t cycles = 0;
    vector<LoadIssue> issues;
};

// Pass 1: Parse CSV lines and populate a temporary map (detect duplicates, missing fields)
static bool ParseLineCSV(const string& line, size_t lineNo,
    string& outNumber, string& outTitle, vector<string>& outPrereqs,
    LoadResultSummary& summary) {
    summary.linesRead++;

    // Skip empty/comment-only lines gracefully.
    string trimmed = trim(line);
    if (trimmed.empty()) return false;

    // Basic CSV split on commas (titles have no commas per project input).
    std::stringstream ss(line);
    string token;
    vector<string> tokens;
    while (getline(ss, token, ',')) tokens.push_back(token);

    // Need at least courseNumber and title.
    if (tokens.size() < 2) {
        summary.issues.push_back({ lineNo, "MissingField",
                                  "Missing course number or title" });
        return false;
    }

    outNumber = NormalizeCourse(tokens[0]);
    outTitle = trim(tokens[1]);
    outPrereqs.clear();

    // Optional prereqs start at index 2 (ignore blanks).
    for (size_t i = 2; i < tokens.size(); ++i) {
        string p = NormalizeCourse(tokens[i]);
        if (!p.empty()) outPrereqs.push_back(p);
    }

    // Basic field checks
    if (outNumber.empty()) {
        summary.issues.push_back({ lineNo, "MissingField", "Empty course number" });
        return false;
    }
    if (outTitle.empty()) {
        summary.issues.push_back({ lineNo, "MissingField",
                                  "Empty course title for " + outNumber });
        return false;
    }
    return true;
}
/* Reviewer note (Pass 1):
   First pass reads/normalizes each CSV line and records precise line numbers
   for any missing fields/duplicates. This makes bad rows easy to track down. 
   */

   // Pass 2A: Validate prereqs exist; strip unknown prereqs; track self-prereqs
static void ValidatePrereqs(unordered_map<string, Course>& temp,
    LoadResultSummary& summary) {
    for (auto& kv : temp) {
        Course& c = kv.second;
        vector<string> keep;
        for (const string& p : c.prereqs) {
            if (p == c.number) {
                summary.selfPrereqs++;
                summary.issues.push_back({ 0, "SelfPrereq",
                                          "Self prerequisite removed: " + c.number });
                continue; // drop self-edge
            }
            if (temp.find(p) == temp.end()) {
                summary.unknownPrereqs++;
                summary.issues.push_back({ 0, "UnknownPrereq",
                                          "Unknown prereq '" + p + "' for " + c.number });
                continue; // drop unknown
            }
            keep.push_back(p);
        }
        c.prereqs.swap(keep);
    }
}
/* Reviewer note (Pass 2A):
   Cleanup pass removes self-edges and unknown prerequisites. It also counts and
   logs what was dropped so the load summary clearly explains the changes. 
   */

   // Pass 2B: Cycle detection (DFS with color marking). Skips courses that are in cycles.
enum Color { WHITE = 0, GRAY = 1, BLACK = 2 };

static bool DFSDetect(const string& u,
    unordered_map<string, Color>& color,
    unordered_map<string, string>& parent,
    const unordered_map<string, Course>& temp,
    vector<string>& cyclePath) {
    color[u] = GRAY;
    auto it = temp.find(u);
    if (it != temp.end()) {
        for (const string& v : it->second.prereqs) {
            if (color[v] == WHITE) {
                parent[v] = u;
                if (DFSDetect(v, color, parent, temp, cyclePath)) return true;
            }
            else if (color[v] == GRAY) {
                // Found back-edge; reconstruct path v -> ... -> u -> v
                cyclePath.clear();
                string x = u;
                cyclePath.push_back(v);
                while (x != v && !x.empty()) {
                    cyclePath.push_back(x);
                    x = parent.count(x) ? parent[x] : string();
                }
                cyclePath.push_back(v);
                std::reverse(cyclePath.begin(), cyclePath.end());
                return true;
            }
        }
    }
    color[u] = BLACK;
    return false;
}
/* Reviewer note (Cycle detection core):
   This is the circular-dependency detector (DFS + WHITE/GRAY/BLACK). A back-edge
   to GRAY means we found a loop; I rebuild the exact path into cyclePath so the
   summary can show something like "A -> B -> C -> A". */

static unordered_set<string> DetectCyclesAndMark(const unordered_map<string, Course>& temp,
    LoadResultSummary& summary) {
    unordered_set<string> inCycle;
    unordered_map<string, Color> color;
    unordered_map<string, string> parent;
    for (const auto& kv : temp) color[kv.first] = WHITE;

    for (const auto& kv : temp) {
        const string& start = kv.first;
        if (color[start] != WHITE) continue;
        vector<string> cyclePath;
        if (DFSDetect(start, color, parent, temp, cyclePath)) {
            summary.cycles++;
            // Mark every node in the cycle; add issue with readable path.
            string pathStr;
            for (size_t i = 0; i < cyclePath.size(); ++i) {
                if (i) pathStr += " -> ";
                pathStr += cyclePath[i];
                inCycle.insert(cyclePath[i]);
            }
            summary.issues.push_back({ 0, "Cycle", "Cycle detected: " + pathStr });
        }
    }
    return inCycle;
}
/* Reviewer note (Pass 2B summary):
   If a cycle is detected, marks all members so they don’t get inserted.
   The load summary prints the readable cycle path for transparency. */

   // Insert validated (and cycle-free) courses into the hash table
static void InsertValidated(const unordered_map<string, Course>& temp,
    const unordered_set<string>& inCycle,
    HashTable& table,
    LoadResultSummary& summary) {
    for (const auto& kv : temp) {
        const string& num = kv.first;
        const Course& c = kv.second;
        if (inCycle.count(num)) continue; // skip cycle members
        if (table.Insert(c)) {
            summary.inserted++;
        }
        else {
            summary.duplicates++; // duplicate in final table (defensive)
        }
    }
}
/* Reviewer note (Insertion gate):
   Final insert step honors the cycle set so only valid, cycle-free courses
   make it into the hash table. Duplicates are guarded as a last resort. */

   // File Loader Orchestrator (multi-pass, timed)
static LoadResultSummary LoadCoursesFromFile(const string& filePath, HashTable& table) {
    LoadResultSummary summary;
    unordered_map<string, Course> temp; // number -> Course

    auto t0 = std::chrono::high_resolution_clock::now();

    ifstream fin(filePath);
    if (!fin.is_open()) {
        summary.issues.push_back({ 0, "FileError", "Cannot open file: " + filePath });
        return summary;
    }

    // Pass 1: parse/normalize; detect duplicates/missing fields with line numbers.
    string line;
    size_t lineNo = 0;
    while (getline(fin, line)) {
        ++lineNo;
        string number, title;
        vector<string> prereqs;
        if (!ParseLineCSV(line, lineNo, number, title, prereqs, summary)) {
            // parsing error already recorded (with line number)
            continue;
        }
        // Duplicate header detection within the same load.
        if (temp.find(number) != temp.end()) {
            summary.duplicates++;
            summary.issues.push_back({ lineNo, "Duplicate", "Duplicate course number: " + number });
            continue;
        }
        Course c;
        c.number = number;
        c.title = title;
        c.prereqs = prereqs;
        temp[number] = std::move(c);
        summary.parsedCourses++;
    }
    fin.close();

    // Pass 2A: prerequisite existence + self-prereq pruning; track unknowns/selfs.
    ValidatePrereqs(temp, summary);

    // Pass 2B: detect cycles and skip cycle members from insertion.
    unordered_set<string> inCycle = DetectCyclesAndMark(temp, summary);

    // Insert into hash table (duplicates guarded).
    InsertValidated(temp, inCycle, table, summary);

    auto t1 = std::chrono::high_resolution_clock::now();
    auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(t1 - t0).count();
    summary.issues.push_back({ 0, "Timing", "Load completed in " + std::to_string(ms) + " ms" });
    return summary;
}
/* Reviewer note (Loader orchestration):
   The loader runs in clear passes (Parse → Validate → Detect cycles → Insert),
   then logs a one-shot summary including timing so performance can be discussed. 
   */

   // Presentation helpers (UI)
static void PrintLoadSummary(const LoadResultSummary& s) {
    cout << "\n=== Load Summary ===\n";
    cout << "Lines read:        " << s.linesRead << "\n";
    cout << "Courses parsed:    " << s.parsedCourses << "\n";
    cout << "Inserted:          " << s.inserted << "\n";
    cout << "Duplicates:        " << s.duplicates << "\n";
    cout << "Unknown prereqs:   " << s.unknownPrereqs << "\n";
    cout << "Self prereqs:      " << s.selfPrereqs << "\n";
    cout << "Cycles detected:   " << s.cycles << "\n";
    for (const auto& issue : s.issues) {
        // Only show detailed validation/timing/cycle logs; lineNo 0 indicates non-line-specific.
        if (issue.type == "Timing")
            cout << "* " << issue.detail << "\n";
        else if (issue.lineNo > 0)
            cout << "* [line " << issue.lineNo << "] " << issue.type << ": " << issue.detail << "\n";
        else
            cout << "* " << issue.type << ": " << issue.detail << "\n";
    }
    cout << "====================\n\n";
}

// Show all courses alphanumerically without mutating the hash table.
static void PrintAll(const HashTable& table) {
    auto t0 = std::chrono::high_resolution_clock::now();
    vector<Course> v = table.ToVectorSorted();
    auto t1 = std::chrono::high_resolution_clock::now();
    auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(t1 - t0).count();

    if (v.empty()) {
        cout << "No courses loaded. Use option 1 to load data first.\n\n";
        return;
    }

    cout << "\nHere is a sample schedule:\n\n";
    for (const Course& c : v) {
        cout << c.number << ", " << c.title << "\n";
    }
    cout << "\n(List generated in " << ms << " ms)\n\n";
}
/* Reviewer note (Listing + timing):
   Listing uses the non-destructive ToVectorSorted() and prints the elapsed time.
   This supports the runtime analysis discussion with actual numbers. */

   // Look up one course and print title + prerequisites with titles.
static void PrintCourse(const HashTable& table, const string& rawInput) {
    string key = NormalizeCourse(rawInput);
    const Course* c = table.Search(key);
    if (!c) {
        cout << "Course not found: " << key << "\n\n";
        return;
    }
    cout << c->number << ", " << c->title << "\n";
    if (c->prereqs.empty()) {
        cout << "Prerequisites: None\n\n";
        return;
    }
    cout << "Prerequisites: ";
    bool first = true;
    for (const string& p : c->prereqs) {
        const Course* pc = table.Search(p);
        if (!first) cout << ", ";
        if (pc) cout << pc->number;
        else    cout << p << " (Not found)";
        first = false;
    }
    cout << "\n";
    // Also print titles under each prereq for clarity (as per directions/sample).
    for (const string& p : c->prereqs) {
        const Course* pc = table.Search(p);
        if (pc) cout << "  - " << pc->number << ": " << pc->title << "\n";
        else    cout << "  - " << p << ": [Title not found]\n";
    }
    cout << "\n";
}

// Robust menu loop with input sanitization and help.
static void MenuLoop() {
    HashTable table;         // main data store
    bool hasLoaded = false;  // gate printing/searching until load occurs

    cout << "Welcome to the course planner.\n\n";

    while (true) {
        cout << "  1. Load Data Structure.\n"
            "  2. Print Course List.\n"
            "  3. Print Course.\n"
            "  9. Exit\n\n"
            "What would you like to do? ";

        string choice;
        if (!getline(cin, choice)) break;
        if (!choice.empty()) choice = trim(choice);

        if (choice == "1") {
            cout << "Enter file name (e.g., courses.txt): ";
            string path;
            if (!getline(cin, path)) break;
            path = trim(path);
            if (path.empty()) {
                cout << "File name cannot be empty.\n\n";
                continue;
            }

            // Load (multi-pass + summary), rebuilds the table on every load for clarity.
            table = HashTable(); // reset
            auto summary = LoadCoursesFromFile(path, table);
            PrintLoadSummary(summary);
            hasLoaded = (summary.inserted > 0);

        }
        else if (choice == "2") {
            if (!hasLoaded) {
                cout << "Please load the data structure first (option 1).\n\n";
                continue;
            }
            PrintAll(table);

        }
        else if (choice == "3") {
            if (!hasLoaded) {
                cout << "Please load the data structure first (option 1).\n\n";
                continue;
            }
            while (true) {
                cout << "What course do you want to know about? (or press Enter to cancel): ";
                string input;
                getline(cin, input);
                input = trim(input);
                if (input.empty()) { cout << "(cancelled)\n\n"; break; }
                PrintCourse(table, input);
                break;
            }

        }
        else if (choice == "9") {
            cout << "Thank you for using the course planner!\n";
            break;

        }
        else if (choice == "h" || choice == "H" || choice == "?") {
            PrintHelp();
        }
        else {
            cout << choice << " is not a valid option.\n\n";
            // Show quick hint to improve UX
            cout << "Try: 1 (Load), 2 (List), 3 (Course), 9 (Exit), or H for help.\n\n";
        }
    }
}
/* Reviewer note (Menu UX):
   Menu accepts H/h/? for contextual help, politely re-prompts on bad input,
   and blocks search/print until a successful load happens. This guards the UX. 
   */

   // Entry Point
int main() {
    MenuLoop();
    return 0;
}

// CS300_Capstone_Enhanced.cpp
// Alondra Paulino Santos
// CS 499 - Algorithms and Data Structures Enhancement
//
// Capstone Course Planning Assistant
// -----------------------------------
// This program enhances the original CS 300 Advising Assistance Program. It
// retains a custom hash table with separate chaining for course storage and
// adds a directed prerequisite graph with Kahn's topological-sort algorithm.
// The generated order places every prerequisite before the courses that depend
// on it. Invalid records, missing prerequisites, self-dependencies, and cycles
// are reported instead of being silently removed from the planning data.
//
// Expected input format:
//   COURSE_NUMBER,COURSE_TITLE[,PREREQUISITE_1,PREREQUISITE_2,...]
//
// The CSV reader supports quoted fields and escaped quotation marks. Course
// codes are normalized by removing spaces and hyphens and converting letters
// to uppercase so inputs such as "cs 300" and "CS-300" match "CS300".

#include <algorithm>
#include <cctype>
#include <chrono>
#include <cstddef>
#include <fstream>
#include <functional>
#include <iomanip>
#include <iostream>
#include <memory>
#include <queue>
#include <sstream>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <utility>
#include <vector>

namespace advising {

using Clock = std::chrono::steady_clock;

// -----------------------------------------------------------------------------
// Domain model and input-normalization helpers
// -----------------------------------------------------------------------------

// A Course is the central record shared by the hash table, search functions,
// and prerequisite graph. Prerequisites are stored as normalized course codes.
struct Course {
    std::string number;
    std::string title;
    std::vector<std::string> prerequisites;
};

std::string Trim(const std::string& value) {
    std::size_t first = 0;
    while (first < value.size() &&
           std::isspace(static_cast<unsigned char>(value[first]))) {
        ++first;
    }

    std::size_t last = value.size();
    while (last > first &&
           std::isspace(static_cast<unsigned char>(value[last - 1]))) {
        --last;
    }
    return value.substr(first, last - first);
}

// Course-code normalization is performed at every input boundary before a
// value is hashed or compared. This keeps file data and user input consistent.
std::string NormalizeCourseCode(const std::string& rawCode) {
    std::string normalized;
    normalized.reserve(rawCode.size());

    for (char character : Trim(rawCode)) {
        const unsigned char byte = static_cast<unsigned char>(character);
        if (std::isspace(byte) || character == '-') {
            continue;
        }
        normalized.push_back(static_cast<char>(std::toupper(byte)));
    }
    return normalized;
}

// SNHU course codes used by this artifact contain an alphabetic department
// prefix followed by three digits, such as CS300, CSCI200, or MATH201.
bool IsValidCourseCode(const std::string& code) {
    // Find the boundary between the department prefix and numeric course level.
    std::size_t prefixLength = 0;
    while (prefixLength < code.size() &&
           std::isalpha(static_cast<unsigned char>(code[prefixLength]))) {
        ++prefixLength;
    }

    if (prefixLength < 2 || prefixLength > 5 ||
        code.size() - prefixLength != 3) {
        return false;
    }

    for (std::size_t index = prefixLength; index < code.size(); ++index) {
        if (!std::isdigit(static_cast<unsigned char>(code[index]))) {
            return false;
        }
    }
    return true;
}

struct Node {
    explicit Node(const Course& courseValue) : course(courseValue) {}

    Course course;
    std::unique_ptr<Node> next;
};

// -----------------------------------------------------------------------------
// Custom hash table with separate chaining
// -----------------------------------------------------------------------------

// The hash table owns each chain through unique_ptr. This prevents leaks and
// double deletion while preserving the separate-chaining design selected in
// the original CS 300 analysis. Copying is disabled; moving and Clear are safe.
class HashTable {
public:
    static constexpr std::size_t kDefaultBucketCount = 179;

    explicit HashTable(std::size_t bucketCount = kDefaultBucketCount)
        : buckets_(std::max<std::size_t>(bucketCount, 1)) {}

    HashTable(const HashTable&) = delete;
    HashTable& operator=(const HashTable&) = delete;
    HashTable(HashTable&&) noexcept = default;
    HashTable& operator=(HashTable&&) noexcept = default;
    ~HashTable() = default;

    bool Insert(const Course& course) {
        const std::size_t bucketIndex = Hash(course.number);

        // Traverse the selected chain before insertion so a duplicate key can
        // never create two records for the same course.
        for (Node* current = buckets_[bucketIndex].get(); current != nullptr;
             current = current->next.get()) {
            if (current->course.number == course.number) {
                return false;
            }
        }

        if (buckets_[bucketIndex] != nullptr) {
            // A nonempty bucket means the new key collided with an existing
            // key. Separate chaining preserves both records in the same bucket.
            ++collisionCount_;
        }

        // Insert at the front of the chain in constant time.
        auto node = std::make_unique<Node>(course);
        node->next = std::move(buckets_[bucketIndex]);
        buckets_[bucketIndex] = std::move(node);
        ++size_;
        return true;
    }

    const Course* Search(const std::string& rawCourseNumber) const {
        const std::string key = NormalizeCourseCode(rawCourseNumber);
        const std::size_t bucketIndex = Hash(key);

        // Only the computed bucket is traversed. With a well-distributed hash,
        // this provides approximately constant-time average lookup.
        for (const Node* current = buckets_[bucketIndex].get(); current != nullptr;
             current = current->next.get()) {
            if (current->course.number == key) {
                return &current->course;
            }
        }
        return nullptr;
    }

    std::vector<Course> ToVector() const {
        std::vector<Course> courses;
        courses.reserve(size_);

        for (const auto& bucket : buckets_) {
            for (const Node* current = bucket.get(); current != nullptr;
                 current = current->next.get()) {
                courses.push_back(current->course);
            }
        }
        return courses;
    }

    std::vector<Course> ToSortedVector() const {
        std::vector<Course> courses = ToVector();

        // Hash tables do not preserve a meaningful display order, so a copied
        // vector is sorted without changing the table's internal chains.
        std::sort(courses.begin(), courses.end(),
                  [](const Course& left, const Course& right) {
                      return left.number < right.number;
                  });
        return courses;
    }

    void Clear() noexcept {
        // Resetting each owning pointer releases the entire linked chain.
        for (auto& bucket : buckets_) {
            bucket.reset();
        }
        size_ = 0;
        collisionCount_ = 0;
    }

    bool Empty() const noexcept { return size_ == 0; }
    std::size_t Size() const noexcept { return size_; }
    std::size_t BucketCount() const noexcept { return buckets_.size(); }
    std::size_t CollisionCount() const noexcept { return collisionCount_; }

    double LoadFactor() const noexcept {
        return static_cast<double>(size_) /
               static_cast<double>(buckets_.size());
    }

    std::size_t MaximumChainLength() const noexcept {
        std::size_t maximum = 0;
        for (const auto& bucket : buckets_) {
            std::size_t length = 0;
            for (const Node* current = bucket.get(); current != nullptr;
                 current = current->next.get()) {
                ++length;
            }
            maximum = std::max(maximum, length);
        }
        return maximum;
    }

private:
    std::size_t Hash(const std::string& key) const {
        // std::hash supplies the hash value; modulo maps it to a valid bucket.
        return std::hash<std::string>{}(key) % buckets_.size();
    }

    std::vector<std::unique_ptr<Node>> buckets_;
    std::size_t size_ = 0;
    std::size_t collisionCount_ = 0;
};

enum class IssueSeverity { kWarning, kError };

struct LoadIssue {
    std::size_t lineNumber = 0;
    IssueSeverity severity = IssueSeverity::kError;
    std::string category;
    std::string detail;
};

struct LoadResult {
    bool fileOpened = false;
    bool hasBlockingRecordErrors = false;
    std::size_t linesRead = 0;
    std::size_t blankLinesSkipped = 0;
    std::size_t parsedCourses = 0;
    std::size_t insertedCourses = 0;
    std::size_t duplicateCourses = 0;
    std::size_t missingPrerequisites = 0;
    std::size_t selfPrerequisites = 0;
    std::size_t duplicatePrerequisites = 0;
    long long elapsedMilliseconds = 0;
    std::vector<LoadIssue> issues;

    bool CanGeneratePlan() const {
        // These conditions cover errors known before topological sorting. A
        // cycle is discovered later when Kahn's algorithm processes the graph.
        return fileOpened && insertedCourses > 0 &&
               !hasBlockingRecordErrors && missingPrerequisites == 0 &&
               selfPrerequisites == 0;
    }
};

// Parses one CSV row while respecting quoted fields and doubled quotes. The
// parser intentionally reports unmatched quotation marks instead of accepting
// an ambiguous record.
bool ParseCsvFields(const std::string& line,
                    std::vector<std::string>& fields,
                    std::string& errorMessage) {
    fields.clear();
    std::string field;
    bool inQuotes = false;

    for (std::size_t index = 0; index < line.size(); ++index) {
        const char character = line[index];

        if (character == '"') {
            // Two consecutive quotation marks inside a quoted value represent
            // one literal quotation mark in CSV.
            if (inQuotes && index + 1 < line.size() && line[index + 1] == '"') {
                field.push_back('"');
                ++index;
            } else {
                inQuotes = !inQuotes;
            }
        } else if (character == ',' && !inQuotes) {
            fields.push_back(Trim(field));
            field.clear();
        } else {
            field.push_back(character);
        }
    }

    if (inQuotes) {
        errorMessage = "Unmatched quotation mark";
        return false;
    }

    fields.push_back(Trim(field));
    return true;
}

bool ParseCourseRecord(const std::string& line,
                       std::size_t lineNumber,
                       Course& course,
                       LoadResult& result) {
    std::vector<std::string> fields;
    std::string parseError;
    if (!ParseCsvFields(line, fields, parseError)) {
        result.hasBlockingRecordErrors = true;
        result.issues.push_back(
            {lineNumber, IssueSeverity::kError, "MalformedCSV", parseError});
        return false;
    }

    if (fields.size() < 2) {
        result.hasBlockingRecordErrors = true;
        result.issues.push_back({lineNumber, IssueSeverity::kError,
                                 "MissingField",
                                 "A course number and title are required"});
        return false;
    }

    course.number = NormalizeCourseCode(fields[0]);
    course.title = Trim(fields[1]);
    course.prerequisites.clear();

    // Invalid primary fields make the entire record unusable because the hash
    // table requires a trustworthy, unique course key.
    if (!IsValidCourseCode(course.number)) {
        result.hasBlockingRecordErrors = true;
        result.issues.push_back(
            {lineNumber, IssueSeverity::kError, "InvalidCourseCode",
             "Invalid course number: " + course.number});
        return false;
    }

    if (course.title.empty()) {
        result.hasBlockingRecordErrors = true;
        result.issues.push_back(
            {lineNumber, IssueSeverity::kError, "MissingField",
             "Course title is empty for " + course.number});
        return false;
    }

    std::unordered_set<std::string> seenPrerequisites;
    for (std::size_t index = 2; index < fields.size(); ++index) {
        const std::string prerequisite = NormalizeCourseCode(fields[index]);
        if (prerequisite.empty()) {
            continue;
        }

        if (!IsValidCourseCode(prerequisite)) {
            result.hasBlockingRecordErrors = true;
            result.issues.push_back(
                {lineNumber, IssueSeverity::kError, "InvalidPrerequisiteCode",
                 "Invalid prerequisite '" + prerequisite + "' for " +
                     course.number});
        }

        if (!seenPrerequisites.insert(prerequisite).second) {
            // Repeated edges would incorrectly increase the prerequisite count
            // during topological sorting, so each relationship is stored once.
            ++result.duplicatePrerequisites;
            result.issues.push_back(
                {lineNumber, IssueSeverity::kWarning, "DuplicatePrerequisite",
                 "Repeated prerequisite '" + prerequisite + "' for " +
                     course.number + " was counted once"});
            continue;
        }
        course.prerequisites.push_back(prerequisite);
    }
    return true;
}

// Relationship validation reports missing and self prerequisites but does not
// delete them. Preserving the original edges prevents the planner from showing
// a misleading course order based on silently repaired data.
void ValidateRelationships(
    const std::unordered_map<std::string, Course>& parsedCourses,
    const std::unordered_map<std::string, std::size_t>& sourceLines,
    LoadResult& result) {
    for (const auto& entry : parsedCourses) {
        const Course& course = entry.second;
        const std::size_t lineNumber = sourceLines.at(course.number);

        for (const std::string& prerequisite : course.prerequisites) {
            if (prerequisite == course.number) {
                ++result.selfPrerequisites;
                result.issues.push_back(
                    {lineNumber, IssueSeverity::kError, "SelfPrerequisite",
                     course.number + " cannot require itself"});
            } else if (parsedCourses.find(prerequisite) == parsedCourses.end()) {
                ++result.missingPrerequisites;
                result.issues.push_back(
                    {lineNumber, IssueSeverity::kError, "MissingPrerequisite",
                     "Prerequisite '" + prerequisite + "' required by " +
                         course.number + " is not defined"});
            }
        }
    }
}

LoadResult LoadCoursesFromFile(const std::string& filePath, HashTable& table) {
    LoadResult result;
    const auto startTime = Clock::now();
    std::ifstream input(filePath);

    if (!input.is_open()) {
        result.issues.push_back({0, IssueSeverity::kError, "FileError",
                                 "Unable to open file: " + filePath});
        return result;
    }
    result.fileOpened = true;

    std::unordered_map<std::string, Course> parsedCourses;
    std::unordered_map<std::string, std::size_t> sourceLines;
    std::string line;
    std::size_t lineNumber = 0;

    while (std::getline(input, line)) {
        ++lineNumber;
        ++result.linesRead;

        const std::string trimmedLine = Trim(line);
        if (trimmedLine.empty() || trimmedLine.front() == '#') {
            ++result.blankLinesSkipped;
            continue;
        }

        Course course;
        if (!ParseCourseRecord(line, lineNumber, course, result)) {
            // The parser has already recorded a line-specific explanation.
            continue;
        }

        // Duplicates are rejected from the temporary collection and marked as
        // blocking because choosing one conflicting definition would be unsafe.
        if (parsedCourses.find(course.number) != parsedCourses.end()) {
            ++result.duplicateCourses;
            result.hasBlockingRecordErrors = true;
            result.issues.push_back(
                {lineNumber, IssueSeverity::kError, "DuplicateCourse",
                 "Duplicate course number: " + course.number});
            continue;
        }

        sourceLines[course.number] = lineNumber;
        parsedCourses.emplace(course.number, std::move(course));
        ++result.parsedCourses;
    }

    // This second pass can validate cross-record relationships only after all
    // course definitions have been collected.
    ValidateRelationships(parsedCourses, sourceLines, result);

    // Load into a fresh table first. Moving the completed table into place
    // avoids mixing records from two files and avoids unsafe pointer copying.
    HashTable replacement(table.BucketCount());
    for (const auto& entry : parsedCourses) {
        if (replacement.Insert(entry.second)) {
            ++result.insertedCourses;
        }
    }
    table = std::move(replacement);

    const auto endTime = Clock::now();
    result.elapsedMilliseconds =
        std::chrono::duration_cast<std::chrono::milliseconds>(endTime - startTime)
            .count();
    return result;
}

struct CourseOrderResult {
    bool success = false;
    std::vector<Course> orderedCourses;
    std::vector<std::string> errors;
    std::vector<std::string> unresolvedCourses;
};

// -----------------------------------------------------------------------------
// Directed-graph construction and topological sorting
// -----------------------------------------------------------------------------

// Builds an adjacency list with edges prerequisite -> dependent course and
// applies Kahn's algorithm. Graph construction and traversal use O(V + E)
// time and space, where V is courses and E is prerequisite relationships.
CourseOrderResult GenerateCourseOrder(const HashTable& table,
                                      const LoadResult& loadResult) {
    CourseOrderResult result;
    if (table.Empty()) {
        result.errors.push_back(
            "Load course data before generating a course order.");
        return result;
    }

    if (loadResult.hasBlockingRecordErrors) {
        result.errors.push_back(
            "Correct malformed or duplicate course records before planning.");
        return result;
    }

    const std::vector<Course> courses = table.ToVector();

    // courseByNumber supports constant-time access to complete course records.
    // dependents is the adjacency list, and prerequisiteCount is the in-degree
    // collection used by Kahn's algorithm.
    std::unordered_map<std::string, const Course*> courseByNumber;
    std::unordered_map<std::string, std::vector<std::string>> dependents;
    std::unordered_map<std::string, std::size_t> prerequisiteCount;

    courseByNumber.reserve(courses.size());
    dependents.reserve(courses.size());
    prerequisiteCount.reserve(courses.size());

    for (const Course& course : courses) {
        // Initialize every vertex even when it has no incoming or outgoing edge.
        courseByNumber[course.number] = &course;
        dependents[course.number] = {};
        prerequisiteCount[course.number] = 0;
    }

    bool relationshipsAreValid = true;
    for (const Course& course : courses) {
        std::unordered_set<std::string> distinctPrerequisites;
        for (const std::string& prerequisite : course.prerequisites) {
            if (!distinctPrerequisites.insert(prerequisite).second) {
                continue;
            }
            if (prerequisite == course.number) {
                relationshipsAreValid = false;
                result.errors.push_back(course.number + " cannot require itself.");
            } else if (courseByNumber.find(prerequisite) == courseByNumber.end()) {
                relationshipsAreValid = false;
                result.errors.push_back("Missing prerequisite " + prerequisite +
                                        " required by " + course.number + ".");
            } else {
                // The edge points from prerequisite to dependent. The dependent
                // cannot enter the ready queue until all incoming edges clear.
                dependents[prerequisite].push_back(course.number);
                ++prerequisiteCount[course.number];
            }
        }
    }

    if (!relationshipsAreValid) {
        result.errors.push_back(
            "No complete course order was generated from invalid relationships.");
        return result;
    }

    std::queue<std::string> readyCourses;
    for (const Course& course : courses) {
        // Courses with zero incoming edges can be taken immediately.
        if (prerequisiteCount[course.number] == 0) {
            readyCourses.push(course.number);
        }
    }

    while (!readyCourses.empty()) {
        // Removing a ready vertex simulates completing that course. Each
        // outgoing edge is then removed from its dependent course.
        const std::string current = readyCourses.front();
        readyCourses.pop();
        result.orderedCourses.push_back(*courseByNumber.at(current));

        for (const std::string& dependent : dependents[current]) {
            std::size_t& remaining = prerequisiteCount[dependent];
            --remaining;
            if (remaining == 0) {
                // All prerequisites for this dependent have now been processed.
                readyCourses.push(dependent);
            }
        }
    }

    if (result.orderedCourses.size() != courses.size()) {
        // Any vertex with a remaining incoming edge belongs to, or depends on,
        // a cycle. A partial result must not be presented as a complete plan.
        for (const Course& course : courses) {
            if (prerequisiteCount[course.number] > 0) {
                result.unresolvedCourses.push_back(course.number);
            }
        }
        std::sort(result.unresolvedCourses.begin(),
                  result.unresolvedCourses.end());
        result.orderedCourses.clear();
        result.errors.push_back(
            "Circular dependencies prevent a complete course order.");
        return result;
    }

    result.success = true;
    return result;
}

void PrintLoadSummary(const LoadResult& result, const HashTable& table) {
    std::cout << "\n=== Load Summary ===\n"
              << "Lines read:             " << result.linesRead << '\n'
              << "Blank/comments skipped: " << result.blankLinesSkipped << '\n'
              << "Courses parsed:         " << result.parsedCourses << '\n'
              << "Courses inserted:       " << result.insertedCourses << '\n'
              << "Duplicate courses:      " << result.duplicateCourses << '\n'
              << "Missing prerequisites:  " << result.missingPrerequisites << '\n'
              << "Self prerequisites:     " << result.selfPrerequisites << '\n'
              << "Duplicate prerequisites:" << result.duplicatePrerequisites << '\n'
              << "Load time:              " << result.elapsedMilliseconds
              << " ms\n";

    if (result.fileOpened) {
        std::cout << "Hash-table load factor: " << std::fixed
                  << std::setprecision(3) << table.LoadFactor() << '\n'
                  << "Hash collisions:        " << table.CollisionCount() << '\n'
                  << "Maximum chain length:   " << table.MaximumChainLength()
                  << '\n';
    }

    if (!result.issues.empty()) {
        std::cout << "\nValidation details:\n";
        for (const LoadIssue& issue : result.issues) {
            std::cout << "- "
                      << (issue.severity == IssueSeverity::kError ? "ERROR" :
                                                                  "WARNING");
            if (issue.lineNumber > 0) {
                std::cout << " [line " << issue.lineNumber << ']';
            }
            std::cout << ' ' << issue.category << ": " << issue.detail << '\n';
        }
    }

    if (result.CanGeneratePlan()) {
        std::cout << "\nRecord and prerequisite-reference validation passed. "
                     "Cycle validation will run when a course order is "
                     "generated.\n";
    } else if (result.fileOpened && result.insertedCourses > 0) {
        std::cout << "\nCourse records are available for review, but validation "
                     "errors must be corrected before generating a plan.\n";
    }
    std::cout << "====================\n\n";
}

void PrintAllCourses(const HashTable& table) {
    if (table.Empty()) {
        std::cout << "No courses are loaded. Use option 1 first.\n\n";
        return;
    }

    const auto startTime = Clock::now();
    const std::vector<Course> courses = table.ToSortedVector();
    const auto endTime = Clock::now();
    const auto elapsed =
        std::chrono::duration_cast<std::chrono::microseconds>(endTime - startTime)
            .count();

    std::cout << "\nCourse Catalog\n--------------\n";
    for (const Course& course : courses) {
        std::cout << course.number << ", " << course.title << '\n';
    }
    std::cout << "\nAlphabetical list generated in " << elapsed
              << " microseconds.\n\n";
}

void PrintCourse(const HashTable& table, const std::string& rawCourseNumber) {
    const std::string courseNumber = NormalizeCourseCode(rawCourseNumber);
    if (!IsValidCourseCode(courseNumber)) {
        std::cout << "Enter a valid course number such as CS300 or CSCI200.\n\n";
        return;
    }

    const Course* course = table.Search(courseNumber);
    if (course == nullptr) {
        std::cout << "Course not found: " << courseNumber << "\n\n";
        return;
    }

    std::cout << '\n' << course->number << ", " << course->title << '\n';
    if (course->prerequisites.empty()) {
        std::cout << "Prerequisites: None\n\n";
        return;
    }

    std::cout << "Prerequisites:\n";
    for (const std::string& prerequisite : course->prerequisites) {
        const Course* prerequisiteCourse = table.Search(prerequisite);
        if (prerequisiteCourse != nullptr) {
            std::cout << "  - " << prerequisiteCourse->number << ": "
                      << prerequisiteCourse->title << '\n';
        } else {
            std::cout << "  - " << prerequisite << ": [course not found]\n";
        }
    }
    std::cout << '\n';
}

void PrintCourseOrder(const HashTable& table, const LoadResult& loadResult) {
    // Timing is informational and helps connect the implementation to the
    // runtime analysis discussed in the enhancement narrative.
    const auto startTime = Clock::now();
    const CourseOrderResult order = GenerateCourseOrder(table, loadResult);
    const auto endTime = Clock::now();
    const auto elapsed =
        std::chrono::duration_cast<std::chrono::microseconds>(endTime - startTime)
            .count();

    if (!order.success) {
        std::cout << "\nUnable to generate a complete course order:\n";
        for (const std::string& error : order.errors) {
            std::cout << "- " << error << '\n';
        }
        if (!order.unresolvedCourses.empty()) {
            std::cout << "Unresolved courses:";
            for (const std::string& course : order.unresolvedCourses) {
                std::cout << ' ' << course;
            }
            std::cout << '\n';
        }
        std::cout << '\n';
        return;
    }

    std::cout << "\nSuggested Prerequisite-Respecting Course Order\n"
              << "-----------------------------------------------\n";
    for (std::size_t index = 0; index < order.orderedCourses.size(); ++index) {
        const Course& course = order.orderedCourses[index];
        std::cout << std::setw(3) << index + 1 << ". " << course.number << ", "
                  << course.title << '\n';
    }
    std::cout << "\nThis order respects prerequisite relationships but does not "
                 "assign semesters or account for course availability or "
                 "credit limits.\n"
              << "Topological order generated in " << elapsed
              << " microseconds.\n\n";
}

    std::cout
        << "\nHelp\n----\n"
        << "1. Load Course Data - Load and validate a CSV course file.\n"
        << "2. Print Course List - Display courses alphabetically.\n"
        << "3. Print Course - Display one course and its prerequisites.\n"
        << "4. Generate Course Order - Place prerequisites before dependents.\n"
        << "9. Exit - Close the application.\n"
        << "Enter H or ? at the menu to display this help.\n\n";
}

void RunApplication() {
    HashTable table;
    LoadResult latestLoad;
    bool loadAttempted = false;

    std::cout << "Welcome to the Capstone Course Planning Assistant.\n";

    while (true) {
        std::cout << "\n  1. Load Course Data\n"
                  << "  2. Print Course List\n"
                  << "  3. Print Course\n"
                  << "  4. Generate Prerequisite-Respecting Course Order\n"
                  << "  9. Exit\n\n"
                  << "What would you like to do? ";

        std::string choice;
        if (!std::getline(std::cin, choice)) {
            std::cout << "\nInput closed. Exiting the application.\n";
            break;
        }
        choice = Trim(choice);

        if (choice == "1") {
            std::cout << "Enter the course data file path: ";
            std::string filePath;
            if (!std::getline(std::cin, filePath)) {
                std::cout << "\nInput closed. Exiting the application.\n";
                break;
            }
            filePath = Trim(filePath);
            if (filePath.empty()) {
                std::cout << "A file path is required.\n";
                continue;
            }

            latestLoad = LoadCoursesFromFile(filePath, table);
            loadAttempted = true;
            PrintLoadSummary(latestLoad, table);
        } else if (choice == "2") {
            PrintAllCourses(table);
        } else if (choice == "3") {
            if (table.Empty()) {
                std::cout << "Load course data before searching.\n";
                continue;
            }
            std::cout << "Enter a course number (or press Enter to cancel): ";
            std::string courseNumber;
            if (!std::getline(std::cin, courseNumber)) {
                std::cout << "\nInput closed. Exiting the application.\n";
                break;
            }
            if (Trim(courseNumber).empty()) {
                std::cout << "Search cancelled.\n";
                continue;
            }
            PrintCourse(table, courseNumber);
        } else if (choice == "4") {
            // Generation is intentionally separate from loading so users can
            // review validation results before requesting an academic plan.
            if (!loadAttempted) {
                std::cout << "Load course data before generating an order.\n";
                continue;
            }
            PrintCourseOrder(table, latestLoad);
        } else if (choice == "9") {
            std::cout << "Thank you for using the Capstone Course Planning "
                         "Assistant.\n";
            break;
        } else if (choice == "H" || choice == "h" || choice == "?") {
            PrintHelp();
        } else {
            std::cout << "'" << choice
                      << "' is not a valid option. Enter H for help.\n";
        }
    }
}

}  // namespace advising

int main() {
    advising::RunApplication();
    return 0;
}

# CS 499 Enhancement Two: Algorithms and Data Structures

This folder contains the original and enhanced versions of the CS 300 course-planning artifact used for the algorithms and data structures category of the CS 499 capstone.

## Enhancement summary

The enhanced application preserves the custom hash table for course storage and lookup, then models prerequisites as a directed graph. It uses Kahn's topological-sort algorithm with an adjacency list, incoming-edge counts, and a queue to generate a prerequisite-respecting course order. Graph construction and traversal require `O(V + E)` time and `O(V + E)` additional space, where `V` is the number of courses and `E` is the number of prerequisite relationships.

The loader also reports duplicate courses, repeated prerequisite edges, malformed records, invalid course codes, missing prerequisites, and self-dependencies. Course-order generation is blocked when invalid relationships could make the recommendation unreliable, and cycles are reported with unresolved course codes.

## Build and run

```bash
clang++ -std=c++17 -Wall -Wextra -Wpedantic -Werror CS300_Capstone_Enhanced.cpp -o cs300_enhanced
./cs300_enhanced
```

Inside the program, select option `1`, enter one of the CSV test-file names, and select option `4` to generate or validate the course order. Select option `9` to exit.
